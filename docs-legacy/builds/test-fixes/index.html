Create your first agent#
 1from haive.agents import SimpleAgent
 2from haive.core.engine import AugLLMConfig
 3
 4# Configure with enhanced settings
 5config = AugLLMConfig(
 6    temperature=0.7,
 7    system_message="You are a helpful AI assistant with expertise in technical topics.",
 8    max_tokens=1000
 9)
10
11# Create and run agent
12agent = SimpleAgent(name="assistant", engine=config)
13response = await agent.arun("Explain quantum computing in simple terms")
14print(response)
Agent with tools#
 1from haive.agents import ReactAgent
 2from haive.tools import WebSearchTool, CalculatorTool
 3
 4# Create tools
