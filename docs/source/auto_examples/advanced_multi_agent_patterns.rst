    import asyncio
    from typing import Dict, Any, List, Optional
    from pydantic import BaseModel, Field

    from haive.agents.simple import SimpleAgent
    from haive.agents.react import ReactAgent
    from haive.agents.multi import MultiAgent
    from haive.agents.multi.base import SequentialAgent, ParallelAgent
    from haive.core.engine.aug_llm import AugLLMConfig
    from haive.core.graph.message import BaseMessage
    from langchain_core.tools import tool


    # === 1. PARALLEL RESEARCH WITH AGGREGATION ===

    class ResearchResult(BaseModel):
        """Structured output for research agents."""
        source: str
        findings: List[str]
        confidence: float
        raw_data: Optional[Dict[str, Any]] = None


    class AggregatedAnalysis(BaseModel):
        """Final aggregated analysis."""
        consensus_findings: List[str]
        conflicting_points: List[Dict[str, Any]]
        confidence_score: float
        recommendations: List[str]
        sources_analyzed: int


    async def parallel_research_pattern():
        """Multiple researchers work in parallel, then results are aggregated."""
    
        # Create specialized research agents
        academic_researcher = SimpleAgent(
            name="academic_researcher",
            engine=AugLLMConfig(temperature=0.3),
            structured_output_model=ResearchResult,
            system_message="You are an academic researcher. Focus on peer-reviewed sources."
        )
    
        industry_researcher = SimpleAgent(
            name="industry_researcher", 
            engine=AugLLMConfig(temperature=0.5),
            structured_output_model=ResearchResult,
            system_message="You are an industry analyst. Focus on market trends and business implications."
        )
    
        technical_researcher = SimpleAgent(
            name="technical_researcher",
            engine=AugLLMConfig(temperature=0.3),
            structured_output_model=ResearchResult,
            system_message="You are a technical expert. Focus on implementation details and technical feasibility."
        )
    
        # Aggregator agent that synthesizes all research
        aggregator = SimpleAgent(
            name="research_aggregator",
            engine=AugLLMConfig(temperature=0.4),
            structured_output_model=AggregatedAnalysis,
            system_message="""You are a research synthesis expert. 
            Analyze multiple research perspectives and create a unified analysis.
            Identify consensus, conflicts, and provide actionable recommendations."""
        )
    
        # Create parallel research phase
        parallel_research = ParallelAgent(
            name="parallel_researchers",
            agents=[academic_researcher, industry_researcher, technical_researcher]
        )
    
        # Sequential: parallel research → aggregation
        research_pipeline = SequentialAgent(
            name="research_pipeline",
            agents=[parallel_research, aggregator]
        )
    
        # Execute
        result = await research_pipeline.arun(
            "Analyze the impact of quantum computing on cryptography"
        )
    
        return result


    # === 2. DYNAMIC ROUTING WITH MULTIPLE BRANCHES ===

    class TaskClassification(BaseModel):
        """Classification result."""
        category: str = Field(..., description="Main category: technical/business/creative/research")
        complexity: str = Field(..., description="Complexity level: simple/moderate/complex")
        subtasks: List[str] = Field(default_factory=list, description="Identified subtasks")
        requires_tools: bool = Field(default=False)


    async def dynamic_routing_pattern():
        """Complex routing based on task classification."""
    
        # Classifier agent
        classifier = SimpleAgent(
            name="task_classifier",
            engine=AugLLMConfig(temperature=0.2),
            structured_output_model=TaskClassification,
            system_message="Classify tasks by type, complexity, and requirements."
        )
    
        # Simple task handlers (no tools)
        simple_tech = SimpleAgent(name="simple_tech", engine=AugLLMConfig())
        simple_biz = SimpleAgent(name="simple_biz", engine=AugLLMConfig())
    
        # Complex task handlers (with tools)
        @tool
        def code_analyzer(code: str) -> str:
            """Analyze code complexity and patterns."""
            return f"Code analysis: {len(code)} chars, complexity score: 0.7"
    
        @tool 
        def market_data(query: str) -> str:
            """Get market data and trends."""
            return f"Market data for {query}: Growing 15% YoY"
    
        complex_tech = ReactAgent(
            name="complex_tech",
            engine=AugLLMConfig(),
            tools=[code_analyzer]
        )
    
        complex_biz = ReactAgent(
            name="complex_biz", 
            engine=AugLLMConfig(),
            tools=[market_data]
        )
    
        # Creative tasks go through a different pipeline
        ideation = SimpleAgent(
            name="ideation",
            engine=AugLLMConfig(temperature=0.9),
            system_message="Generate creative ideas and possibilities."
        )
    
        refinement = SimpleAgent(
            name="refinement",
            engine=AugLLMConfig(temperature=0.5),
            system_message="Refine and structure creative ideas."
        )
    
        creative_pipeline = SequentialAgent(
            name="creative_pipeline",
            agents=[ideation, refinement]
        )
    
        # Research tasks use parallel pattern
        research_pipeline = parallel_research_pattern()  # Reuse from above
    
        # Build the dynamic multi-agent system
        class DynamicRouter(MultiAgent):
            """Custom multi-agent with complex routing logic."""
        
            def __init__(self):
                agents = {
                    "classifier": classifier,
                    "simple_tech": simple_tech,
                    "simple_biz": simple_biz,
                    "complex_tech": complex_tech,
                    "complex_biz": complex_biz,
                    "creative_pipeline": creative_pipeline,
                    "research_pipeline": research_pipeline
                }
            
                super().__init__(
                    name="dynamic_router",
                    agents=agents,
                    entry_point="classifier"
                )
            
                # Add routing logic
                self._add_dynamic_routes()
        
            def _add_dynamic_routes(self):
                """Add complex conditional routing."""
            
                def route_by_classification(state: Dict[str, Any]) -> str:
                    """Route based on classification results."""
                    classification = state.get("task_classification", {})
                    category = classification.get("category", "")
                    complexity = classification.get("complexity", "")
                
                    # Routing decision tree
                    if category == "creative":
                        return "creative_pipeline"
                    elif category == "research":
                        return "research_pipeline"
                    elif category == "technical":
                        return "complex_tech" if complexity == "complex" else "simple_tech"
                    elif category == "business":
                        return "complex_biz" if complexity == "complex" else "simple_biz"
                    else:
                        return "simple_tech"  # Default
            
                self.add_conditional_edges(
                    source="classifier",
                    path=route_by_classification
                )
    
        # Use the dynamic router
        router = DynamicRouter()
        result = await router.arun("Create a technical analysis of blockchain scalability solutions")
    
        return result


    # === 3. MAP-REDUCE PATTERN FOR DOCUMENT PROCESSING ===

    class DocumentChunk(BaseModel):
        """Processed document chunk."""
        chunk_id: int
        summary: str
        key_points: List[str]
        entities: List[str]


    class DocumentSynthesis(BaseModel):
        """Final document synthesis."""
        executive_summary: str
        main_themes: List[str]
        key_insights: List[str]
        action_items: List[str]
        total_chunks_processed: int


    async def map_reduce_pattern():
        """Process large documents using map-reduce pattern."""
    
        # Mapper agents (process chunks in parallel)
        chunk_processors = []
        for i in range(4):  # 4 parallel processors
            processor = SimpleAgent(
                name=f"chunk_processor_{i}",
                engine=AugLLMConfig(temperature=0.3),
                structured_output_model=DocumentChunk,
                system_message=f"You are processor {i}. Extract key information from document chunks."
            )
            chunk_processors.append(processor)
    
        # Parallel chunk processing
        mapper = ParallelAgent(
            name="chunk_mappers",
            agents=chunk_processors
        )
    
        # Reducer agent (synthesize all chunks)
        reducer = SimpleAgent(
            name="document_synthesizer",
            engine=AugLLMConfig(temperature=0.4),
            structured_output_model=DocumentSynthesis,
            system_message="Synthesize information from multiple document chunks into cohesive insights."
        )
    
        # Map-reduce pipeline
        map_reduce = SequentialAgent(
            name="map_reduce_processor",
            agents=[mapper, reducer]
        )
    
        # Simulate chunked document
        document_chunks = [
            "Chapter 1: Introduction to quantum computing basics...",
            "Chapter 2: Quantum algorithms and their applications...",
            "Chapter 3: Current limitations and challenges...",
            "Chapter 4: Future prospects and research directions..."
        ]
    
        result = await map_reduce.arun(
            f"Process these document chunks: {document_chunks}"
        )
    
        return result


    # === 4. HIERARCHICAL DECISION TREE ===

    class StrategicDecision(BaseModel):
        """High-level strategic decision."""
        strategy: str
        priorities: List[str]
        resource_allocation: Dict[str, float]


    class TacticalPlan(BaseModel):
        """Tactical implementation plan."""
        tactical_goals: List[str]
        timeline: str
        dependencies: List[str]


    class OperationalTasks(BaseModel):
        """Operational task list."""
        immediate_actions: List[str]
        assigned_teams: Dict[str, List[str]]
        success_metrics: List[str]


    async def hierarchical_decision_pattern():
        """Three-tier hierarchical decision-making system."""
    
        # Strategic level (CEO)
        strategic_agent = SimpleAgent(
            name="strategic_planner",
            engine=AugLLMConfig(temperature=0.6),
            structured_output_model=StrategicDecision,
            system_message="You are a strategic planner. Make high-level decisions."
        )
    
        # Tactical level (Department heads) - Parallel
        product_tactical = SimpleAgent(
            name="product_tactician",
            engine=AugLLMConfig(temperature=0.5),
            structured_output_model=TacticalPlan,
            system_message="You are a product manager. Create tactical product plans."
        )
    
        engineering_tactical = SimpleAgent(
            name="engineering_tactician",
            engine=AugLLMConfig(temperature=0.5),
            structured_output_model=TacticalPlan,
            system_message="You are an engineering manager. Create tactical tech plans."
        )
    
        marketing_tactical = SimpleAgent(
            name="marketing_tactician",
            engine=AugLLMConfig(temperature=0.5),
            structured_output_model=TacticalPlan,
            system_message="You are a marketing manager. Create tactical marketing plans."
        )
    
        # Parallel tactical planning
        tactical_team = ParallelAgent(
            name="tactical_planners",
            agents=[product_tactical, engineering_tactical, marketing_tactical]
        )
    
        # Operational level (Team leads) - Process each tactical plan
        operational_agent = SimpleAgent(
            name="operational_planner",
            engine=AugLLMConfig(temperature=0.4),
            structured_output_model=OperationalTasks,
            system_message="Convert tactical plans into concrete operational tasks."
        )
    
        # Three-tier hierarchy
        hierarchy = SequentialAgent(
            name="corporate_hierarchy",
            agents=[strategic_agent, tactical_team, operational_agent]
        )
    
        result = await hierarchy.arun(
            "Plan the launch of a new AI-powered product line"
        )
    
        return result


    # === 5. CONSENSUS BUILDING WITH DEBATE ===

    class Argument(BaseModel):
        """Structured argument."""
        position: str
        supporting_points: List[str]
        counter_arguments: List[str]
        confidence: float


    class Consensus(BaseModel):
        """Final consensus."""
        agreed_points: List[str]
        disagreements: List[Dict[str, str]]
        final_recommendation: str
        confidence_level: float


    async def consensus_pattern():
        """Multiple agents debate and reach consensus."""
    
        # Create agents with different perspectives
        optimist = SimpleAgent(
            name="optimist",
            engine=AugLLMConfig(temperature=0.7),
            structured_output_model=Argument,
            system_message="You are an optimistic analyst. See opportunities and positive outcomes."
        )
    
        pessimist = SimpleAgent(
            name="pessimist",
            engine=AugLLMConfig(temperature=0.7),
            structured_output_model=Argument,
            system_message="You are a risk analyst. Identify problems and potential failures."
        )
    
        realist = SimpleAgent(
            name="realist",
            engine=AugLLMConfig(temperature=0.5),
            structured_output_model=Argument,
            system_message="You are a pragmatic analyst. Balance opportunities with constraints."
        )
    
        # Parallel analysis
        analysts = ParallelAgent(
            name="analyst_panel",
            agents=[optimist, pessimist, realist]
        )
    
        # Moderator to build consensus
        moderator = SimpleAgent(
            name="moderator",
            engine=AugLLMConfig(temperature=0.4),
            structured_output_model=Consensus,
            system_message="You are a skilled moderator. Synthesize different viewpoints into consensus."
        )
    
        # Debate → Consensus pipeline
        consensus_builder = SequentialAgent(
            name="consensus_pipeline",
            agents=[analysts, moderator]
        )
    
        result = await consensus_builder.arun(
            "Should we invest in developing a quantum computing division?"
        )
    
        return result


    # === MAIN EXECUTION ===

    async def main():
        """Run all advanced patterns."""
    
        print("🚀 Advanced Multi-Agent Patterns Demo\n")

    if __name__ == "__main__":
